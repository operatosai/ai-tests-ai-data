          if (addScore(2)) return;

    }

  function drawHazard(h){
    ctx.save();
    if (h.life !== undefined && h.life < 1) ctx.globalAlpha = Math.max(0, h.life) * (0.6 + 0.4*Math.sin(h.life*40));
    ctx.translate(h.x, h.y); ctx.rotate(h.rot);
    ctx.shadowColor = h.homing ? '#f0f' : '#f33'; ctx.shadowBlur = 15;
    ctx.fillStyle = h.homing ? '#c2f' : '#e22';
    ctx.beginPath();
    const spikes = 8;
    for (let i = 0; i < spikes*2; i++) {
      const rr = i % 2 === 0 ? h.r : h.r*0.55;
      const a = i * Math.PI / spikes;
      ctx.lineTo(Math.cos(a)*rr, Math.sin(a)*rr);
    }
    ctx.closePath(); ctx.fill();
    ctx.fillStyle = '#300';
    ctx.beginPath(); ctx.arc(0,0,h.r*0.25,0,Math.PI*2); ctx.fill();
    ctx.restore();

    if (blink) return;

    for (let i = 0; i < trail.length; i++) {
