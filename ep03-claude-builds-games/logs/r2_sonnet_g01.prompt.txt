<canvas id="c" width="800" height="600"></canvas>
<script>
(() => {
  const cv = document.getElementById('c'), ctx = cv.getContext('2d');
  const W = cv.width, H = cv.height;
  const WIN_SCORE = 40;
  window.gameState = { score: 0, state: "menu" };

  let state = "menu", score = 0, lives = 3, time = 0, last = 0;
  let player, orbs, hazards, particles, spawnT, invuln, keys = {}, mouse = null, shake = 0;

  // ---------- audio ----------
  let actx = null;
  function beep(freq, dur, type = 'sine', vol = 0.12, slide = 0) {
    try {
      if (!actx) actx = new (window.AudioContext || window.webkitAudioContext)();
      if (actx.state === 'suspended') actx.resume();
      const o = actx.createOscillator(), g = actx.createGain();
      o.type = type; o.frequency.value = freq;
      if (slide) o.frequency.linearRampToValueAtTime(freq + slide, actx.currentTime + dur);
      g.gain.setValueAtTime(vol, actx.currentTime);
      g.gain.exponentialRampToValueAtTime(0.001, actx.currentTime + dur);
      o.connect(g); g.connect(actx.destination);
      o.start(); o.stop(actx.currentTime + dur);
    } catch (e) {}
  }

  function sync() { window.gameState = { score: score, state: state }; }

  function reset() {
    player = { x: W / 2, y: H / 2, r: 14, speed: 300 };
    orbs = []; hazards = []; particles = [];
    score = 0; lives = 3; time = 0; spawnT = 0; invuln = 0; shake = 0;
    for (let i = 0; i < 3; i++) spawnOrb();
    for (let i = 0; i < 3; i++) spawnHazard();
  }

  function start() {
    reset(); state = "playing"; sync();
    beep(440, 0.15, 'square', 0.08, 220);
  }

  function spawnOrb() {
    let x, y, tries = 0;
    do {
      x = 40 + Math.random() * (W - 80); y = 60 + Math.random() * (H - 100); tries++;
    } while (tries < 20 && Math.hypot(x - player.x, y - player.y) < 80);
    orbs.push({ x, y, r: 10, ph: Math.random() * 6 });
  }

  function spawnHazard() {
    const diff = 1 + time / 20;
    const sp = (70 + Math.random() * 50) * Math.min(diff, 3.5);
    let x, y;
    const side = Math.floor(Math.random() * 4);
    if (side === 0) { x = -20; y = Math.random() * H; }
    else if (side === 1) { x = W + 20; y = Math.random() * H; }
    else if (side === 2) { x = Math.random() * W; y = -20; }
    else { x = Math.random() * W; y = H + 20; }
    const a = Math.atan2(player.y - y, player.x - x) + (Math.random() - 0.5) * 1.2;
    hazards.push({ x, y, vx: Math.cos(a) * sp, vy: Math.sin(a) * sp, r: 12 + Math.random() * 8, rot: 0 });
  }

  function burst(x, y, color, n) {
    for (let i = 0; i < n; i++) {
      const a = Math.random() * 6.283, s = 40 + Math.random() * 160;
      particles.push({ x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s, life: 0.5 + Math.random() * 0.4, max: 0.9, color });
    }
  }

  // ---------- input ----------
  addEventListener('keydown', e => {
    keys[e.key.toLowerCase()] = true;
    if (['arrowup', 'arrowdown', 'arrowleft', 'arrowright', ' '].includes(e.key.toLowerCase())) e.preventDefault();
    if (state !== "playing" && (e.key === ' ' || e.key === 'Enter')) start();
    mouse = null;
  });
  addEventListener('keyup', e => { keys[e.key.toLowerCase()] = false; });
  function mpos(e) {
    const r = cv.getBoundingClientRect();
    return { x: (e.clientX - r.left) * W / r.width, y: (e.clientY - r.top) * H / r.height };
  }
  cv.addEventListener('mousemove', e => { mouse = mpos(e); });
  cv.addEventListener('mousedown', e => {
    if (state !== "playing") start();
    mouse = mpos(e);
  });

  // ---------- update ----------
  function update(dt) {
    if (state !== "playing") { updateParticles(dt); return; }
    time += dt;
    invuln = Math.max(0, invuln - dt);
    shake = Math.max(0, shake - dt * 30);

    let dx = 0, dy = 0;
    if (keys['arrowleft'] || keys['a']) dx -= 1;
    if (keys['arrowright'] || keys['d']) dx += 1;
    if (keys['arrowup'] || keys['w']) dy -= 1;
    if (keys['arrowdown'] || keys['s']) dy += 1;
    if (dx || dy) {
      const l = Math.hypot(dx, dy);
      player.x += dx / l * player.speed * dt; player.y += dy / l * player.speed * dt;
    } else if (mouse) {
      const ddx = mouse.x - player.x, ddy = mouse.y - player.y, d = Math.hypot(ddx, ddy);
      const step = Math.min(d, player.speed * 1.5 * dt);
      if (d > 1) { player.x += ddx / d * step; player.y += ddy / d * step; }
    }
    player.x = Math.max(player.r, Math.min(W - player.r, player.x));
    player.y = Math.max(player.r, Math.min(H - player.r, player.y));

    // difficulty: more hazards over time
    spawnT += dt;
    const interval = Math.max(0.7, 3.5 - time * 0.04);
    if (spawnT > interval) { spawnT = 0; spawnHazard(); }

    for (const o of orbs) o.ph += dt * 4;
    for (let i = orbs.length - 1; i >= 0; i--) {
      const o = orbs[i];
      if (Math.hypot(o.x - player.x, o.y - player.y) < o.r + player.r) {
        orbs.splice(i, 1); score++;
        burst(o.x, o.y, '#6ff', 14);
        beep(660 + Math.min(score, 20) * 20, 0.12, 'sine', 0.12, 300);
        spawnOrb();
        if (score % 5 === 0) spawnOrb();
        if (orbs.length > 5) orbs.pop();
        if (score >= WIN_SCORE) { state = "won"; beep(880, 0.5, 'triangle', 0.15, 400); }
        sync();
      }
    }

    for (let i = hazards.length - 1; i >= 0; i--) {
      const h = hazards[i];
      h.x += h.vx * dt; h.y += h.vy * dt; h.rot += dt * 3;
      // bounce off walls
      if (h.x < h.r && h.vx < 0 || h.x > W - h.r && h.vx > 0) h.vx *= -1;
      if (h.y < h.r && h.vy < 0 || h.y > H - h.r && h.vy > 0) h.vy *= -1;
      if (invuln <= 0 && Math.hypot(h.x - player.x, h.y - player.y) < h.r * 0.85 + player.r * 0.8) {
        lives--; invuln = 1.5; shake = 10;
        burst(player.x, player.y, '#f55', 25);
        beep(120, 0.4, 'sawtooth', 0.15, -80);
        if (lives <= 0) { state = "over"; sync(); }
      }
    }
    updateParticles(dt);
  }

  function updateParticles(dt) {
    for (let i = particles.length - 1; i >= 0; i--) {
      const p = particles[i];
      p.x += p.vx * dt; p.y += p.vy * dt; p.life -= dt;
      if (p.life <= 0) particles.splice(i, 1);
    }
  }

  // ---------- draw ----------
  function glowCircle(x, y, r, color, blur) {
    ctx.shadowColor = color; ctx.shadowBlur = blur;
    ctx.fillStyle = color;
    ctx.beginPath(); ctx.arc(x, y, r, 0, 6.283); ctx.fill();
    ctx.shadowBlur = 0;
  }

  function text(t, x, y, size, color, align = 'center') {
    ctx.font = `bold ${size}px 'Courier New', monospace`;
    ctx.fillStyle = color; ctx.textAlign = align;
    ctx.fillText(t, x, y);
  }

  function draw() {
    ctx.save();
    ctx.clearRect(0, 0, W, H);
    if (shake > 0) ctx.translate((Math.random() - 0.5) * shake, (Math.random() - 0.5) * shake);

    if (state !== "menu") {
      for (const o of orbs) glowCircle(o.x, o.y, o.r + Math.sin(o.ph) * 2, '#5ff', 25);
      for (const h of hazards) {
        ctx.save(); ctx.translate(h.x, h.y); ctx.rotate(h.rot);
        ctx.shadowColor = '#f33'; ctx.shadowBlur = 15; ctx.fillStyle = '#e22';
        ctx.beginPath();
        for (let i = 0; i < 16; i++) {
          const rr = i % 2 ? h.r * 0.6 : h.r * 1.15, a = i / 16 * 6.283;
          ctx.lineTo(Math.cos(a) * rr, Math.sin(a) * rr);
        }
        ctx.closePath(); ctx.fill(); ctx.restore();
      }
      if (state !== "over" && !(invuln > 0 && Math.floor(invuln * 12) % 2)) glowCircle(player.x, player.y, player.r, '#ffd84a', 22);
      for (const p of particles) {
        ctx.globalAlpha = Math.max(0, p.life / p.max);
        glowCircle(p.x, p.y, 3, p.color, 8);
      }
      ctx.globalAlpha = 1;

      text('SCORE ' + score + ' / ' + WIN_SCORE, 16, 30, 22, '#fff', 'left');
      text('LIVES ' + '♥'.repeat(Math.max(0, lives)), W - 16, 30, 22, '#f66', 'right');
      const lvl = 1 + Math.floor(time / 15);
      text('LEVEL ' + lvl, W / 2, 30, 18, '#9af');
    }

    if (state === "menu") {
      text('ORB RUNNER', W / 2, 190, 64, '#5ff');
      glowCircle(W / 2 - 150, 240, 8, '#5ff', 20);
      text('Collect glowing orbs', W / 2 + 10, 246, 20, '#cfe');
      ctx.save(); ctx.translate(W / 2 - 150, 280); ctx.fillStyle = '#e22'; ctx.shadowColor = '#f33'; ctx.shadowBlur = 12;
      ctx.beginPath(); for (let i = 0; i < 16; i++) { const rr = i % 2 ? 6 : 12; const a = i / 16 * 6.283; ctx.lineTo(Math.cos(a) * rr, Math.sin(a) * rr); }
      ctx.closePath(); ctx.fill(); ctx.restore();
      text('Avoid the red spikes', W / 2 + 10, 286, 20, '#fcc');
      text('Move: WASD / Arrow keys, or the mouse', W / 2, 345, 20, '#fff');
      text('Reach ' + WIN_SCORE + ' orbs to win. You have 3 lives.', W / 2, 375, 18, '#9af');
      text('Click or press SPACE to start', W / 2, 450, 26, '#ffd84a');
    } else if (state === "over") {
      ctx.fillStyle = 'rgba(0,0,0,0.6)'; ctx.fillRect(0, 0, W, H);
      text('GAME OVER', W / 2, 260, 60, '#f55');
      text('Score: ' + score, W / 2, 315, 28, '#fff');
      text('Click or press SPACE to play again', W / 2, 390, 22, '#ffd84a');
    } else if (state === "won") {
      ctx.fillStyle = 'rgba(0,0,0,0.6)'; ctx.fillRect(0, 0, W, H);
      text('YOU WIN!', W / 2, 260, 60, '#5f8');
      text('Score: ' + score, W / 2, 315, 28, '#fff');
      text('Click or press SPACE to play again', W / 2, 390, 22, '#ffd84a');
    }
    ctx.restore();
  }

  function loop(t) {
    const dt = Math.min(0.05, (t - last) / 1000 || 0);
    last = t;
    update(dt);
    draw();
    requestAnimationFrame(loop);
  }
  reset();
  sync();
  requestAnimationFrame(loop);
})();
