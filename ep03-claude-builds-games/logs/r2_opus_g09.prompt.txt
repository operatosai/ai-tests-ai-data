    if (lives === 1) {
      heartT -= dt;
      if (heartT <= 0) { heartT = 0.95; sfx.heart(); }
    }

    if (comboTimer > 0) { comboTimer -= dt; if (comboTimer <= 0) combo = 0; }

    const magR = MAGNET * (1 + (mult() - 1) * 0.3);
    for (let i = orbs.length-1; i >= 0; i--) {
      const o = orbs[i];
      o.t += dt*3; o.life -= dt; o.born += dt;
      let od = dist(o, player);
      if (od < magR && od > 0.01 && o.born > 0.15) {
        const pull = 160 * (1 - od / magR) * dt * (1 + (mult()-1)*0.2);
        o.x += (player.x - o.x) / od * pull; o.y += (player.y - o.y) / od * pull;
        od = dist(o, player);
      }
      if (od < o.r + player.r + 2 && o.born > 0) {
        const prevM = mult();
        combo++; comboTimer = 2;
        const m = mult();
        let pts = (o.big ? 50 : 10) * m;
        burst(o.x, o.y, o.hue, o.big ? 24 : 12, 180);
        popup(o.x, o.y - 10, '+' + pts + (m > 1 ? ' x' + m : ''), o.big ? '#ffe36b' : `hsl(${o.hue},100%,80%)`, o.big ? 20 : 15);
        if (o.big) sfx.big(); else sfx.orb(combo);
        if (o.bonus) {
          showerGot++;
          if (showerGot === 6) {
            pts += 50; showerClears++;
            popup(player.x, player.y - 56, 'SHOWER CLEAR! +50', '#ffe36b', 22);
            ring(player.x, player.y, '255,227,107', 520, 0.6);
            burst(player.x, player.y, 50, 30, 260);
            setTimeout(sfx.clear, 60);
          }
        }
        if (m > prevM) {
          popup(player.x, player.y - 40, m === 4 ? 'x4 FEVER!' : 'x' + m + ' MULTIPLIER!', m === 4 ? '#f8f' : '#ff6', 22);
          ring(player.x, player.y, m === 4 ? '255,120,255' : '255,255,110', 380, 0.4);
          setTimeout(sfx.multi, 90);
        }
        orbs.splice(i,1);
        if (!o.bonus) spawnOrb();
        if (addScore(pts)) return;
      } else if (o.life <= 0) {
        burst(o.x, o.y, o.hue, 6, 60);
        orbs.splice(i,1);
        if (!o.bonus) spawnOrb();
      }
    }

    // power-ups: shield, or +1 life when hurt
    powerTimer -= dt;
    if (!power && powerTimer <= 0) {
      const kind = (lives < 3 && level >= 2 && Math.random() < 0.4) ? 'life' : 'shield';
      power = { x: rand(60, W-60), y: rand(TOP + 40, H-60), life: 7, t: 0, kind };
      powerTimer = rand(16, 24);
    }
    if (power) {
      power.life -= dt; power.t += dt;
      if (dist(power, player) < player.r + 14) {
        if (power.kind === 'life') {
          lives = Math.min(3, lives + 1); sfx.life();
          popup(power.x, power.y - 14, '+1 LIFE', '#7f9', 20);
          burst(power.x, power.y, 130, 30, 220);
          ring(power.x, power.y, '120,255,150', 300, 0.4);
        } else {
          shield = 5; sfx.power();
          popup(power.x, power.y - 14, 'SHIELD!', '#ffe36b', 20);
          burst(power.x, power.y, 50, 30, 220);
          ring(power.x, power.y, '255,227,107', 300, 0.4);
        }
        power = null;
      } else if (power.life <= 0) power = null;
    }

    if (level >= 4) {
      laserTimer -= dt;
      if (laserTimer <= 0) {
        const h = spawnLaser(0);
        if (level >= 7 && Math.random() < 0.4) {
          const l2 = { horiz: !h, pos: !h ? clamp(player.y + rand(-80,80), TOP + 14, H-20) : clamp(player.x + rand(-100,100), 20, W-20), warn: 1.1, on: 0.35, w: 20, done: false };
          lasers.push(l2);
        }
        sfx.charge();
        laserTimer = Math.max(2.4, 7.5 - (level-4)*0.6) + rand(0, 1.5);
      }
    }
    for (let i = lasers.length-1; i >= 0; i--) {
      const l = lasers[i];
      if (l.warn > 0) {
        l.warn -= dt;
        if (l.warn <= 0) { sfx.zap(); shake = Math.max(shake, 4); }
        continue;
      }
      l.on -= dt;
      const dd = l.horiz ? Math.abs(player.y - l.pos) : Math.abs(player.x - l.pos);
      if (!l.done && dd < l.w/2 + player.r*0.7) {
        if (shield > 0) { l.done = true; popup(player.x, player.y - 20, 'BLOCKED', '#ffe36b', 16); }
        else if (dashT > 0) { l.done = true; sfx.dodge(); popup(player.x, player.y - 20, '+5 DODGE', '#9ff', 15); if (addScore(5)) return; }
        else if (invuln <= 0) { l.done = true; if (hurt('Fried by a laser')) return; }
      }
      if (l.on <= 0) lasers.splice(i, 1);
    }

    for (let i = pending.length-1; i >= 0; i--) {
      pending[i].t -= dt;
      if (pending[i].t <= 0) { spawnHazardAt(pending[i]); pending.splice(i,1); }
    }
    const target = 3 + Math.floor(elapsed / 5) + (still > IDLE_LIMIT ? 2 : 0);
    spawnTimer -= dt;
    if (hazards.length + pending.length < Math.min(target, 24) && spawnTimer <= 0) {
      queueHazard();
      spawnTimer = Math.max(0.45, 1.7 - level*0.15);
    }

    for (let i = hazards.length-1; i >= 0; i--) {
      const h = hazards[i];
      h.life -= dt;
      if (h.life <= 0) { burst(h.x, h.y, h.dart ? 30 : (h.homing ? 290 : 0), 8, 80); hazards.splice(i,1); continue; }
      if (h.homing) {
        const a = Math.atan2(player.y - h.y, player.x - h.x);
        h.vx += Math.cos(a) * 60 * dt; h.vy += Math.sin(a) * 60 * dt;
        const s = Math.hypot(h.vx, h.vy), max = h.speed * 0.9;
        if (s > max) { h.vx *= max/s; h.vy *= max/s; }
      }
      h.x += h.vx*dt; h.y += h.vy*dt; h.rot += h.spin*dt;
      const inside = h.x > h.r && h.x < W-h.r && h.y > TOP + h.r && h.y < H-h.r;
      if (h.dart) {
        if (h.entering) { if (inside) h.entering = false; }
        else if (h.x < -h.r*2 || h.x > W + h.r*2 || h.y < TOP - h.r*2 || h.y > H + h.r*2) { hazards.splice(i,1); continue; }
      } else if (h.entering) {
        if (inside) h.entering = false;
      } else {
        if (h.x < h.r) { h.x = h.r; h.vx = Math.abs(h.vx); }
        if (h.x > W-h.r) { h.x = W-h.r; h.vx = -Math.abs(h.vx); }
        if (h.y < TOP + h.r) { h.y = TOP + h.r; h.vy = Math.abs(h.vy); }
        if (h.y > H-h.r) { h.y = H-h.r; h.vy = -Math.abs(h.vy); }
      }
      if (h.life < 0.5) continue;
      const hd = dist(h, player), hitR = h.r*0.8 + player.r;
      if (hd < hitR) {
        if (shield > 0) {
          hazards.splice(i,1);
          popup(h.x, h.y - 10, '+25 SMASH', '#ffb', 16);
          burst(h.x, h.y, h.dart ? 30 : (h.homing ? 290 : 0), 22, 240);
          sfx.smash(); shake = 5;
          if (addScore(25)) return;
          continue;
        }
        if (dashT > 0) {
          if (!h.dodged) {
            h.dodged = true; h.grazed = true; sfx.dodge();
            popup(h.x, h.y - 12, '+5 DODGE', '#9ff', 15);
            if (addScore(5)) return;
          }
          continue;
        }
        if (safe) continue;
        if (hurt(h.dart ? 'Pierced by a dart' : h.homing ? 'Caught by a hunter' : 'Shredded by a spike mine')) return;
      } else if (hd < hitR + GRAZE) {
        if (!h.grazed && !safe && shield <= 0) {
          h.grazed = true; grazes++;
          sfx.graze();
          const mx = (h.x + player.x) / 2, my = (h.y + player.y) / 2;
          burst(mx, my, 190, 6, 120);
          popup(player.x, player.y - 22, 'GRAZE +2', '#cff', 16);
          if (combo > 0) comboTimer = Math.min(2, comboTimer + 0.6);
          if (addScore(2)) return;
        }
      } else if (hd > hitR + GRAZE + 40) {
        h.grazed = false; h.dodged = false;
      }
    }
  }

  // ---------- Draw ----------
  function glowCircle(x, y, r, color, blur){
    ctx.save();
    ctx.shadowColor = color; ctx.shadowBlur = blur;
    ctx.fillStyle = color;
    ctx.beginPath(); ctx.arc(x, y, Math.max(0.5, r), 0, Math.PI*2); ctx.fill();
    ctx.restore();
  }
  function drawStars(t){
    const px = (player && state !== 'menu') ? (player.x - W/2) : 0;
    const py = (player && state !== 'menu') ? (player.y - H/2) : 0;
    for (const s of stars) {
      let x = s.x - px*s.z*0.06 - t*8*s.z;
      let y = s.y - py*s.z*0.06 + t*3*s.z;
      x = ((x % W) + W) % W; y = ((y % H) + H) % H;
      const a = (0.2 + 0.5*s.z) * (0.6 + 0.4*Math.sin(t*2 + s.tw));
      ctx.fillStyle = `rgba(200,220,255,${a})`;
      const sz = 0.8 + s.z*1.6;
      ctx.fillRect(x, y, sz, sz);
    }
  }
  function drawGrid(t){
    ctx.save();
    const hue = (210 + ((level || 1) - 1) * 32) % 360;
    ctx.strokeStyle = `hsla(${hue},70%,55%,0.08)`; ctx.lineWidth = 1;
    const off = (t*(10 + ((level||1)-1)*2)) % 40;
    for (let x = -40 + off; x < W; x += 40) { ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,H); ctx.stroke(); }
    for (let y = -40 + off; y < H; y += 40) { ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(W,y); ctx.stroke(); }
    if (state !== 'menu') {
      const g = ctx.createRadialGradient(W/2, H/2, 50, W/2, H/2, H*0.8);
      g.addColorStop(0, `hsla(${hue},80%,40%,0.08)`);
      g.addColorStop(1, 'rgba(0,0,0,0)');
      ctx.fillStyle = g; ctx.fillRect(0, 0, W, H);
    }
    ctx.restore();
  }
  function drawHazard(h){
    ctx.save();
    if (h.life !== undefined && h.life < 1) ctx.globalAlpha = Math.max(0, h.life) * (0.6 + 0.4*Math.sin(h.life*40));
    ctx.translate(h.x, h.y);
    if (h.dart) {
      ctx.rotate(Math.atan2(h.vy, h.vx));
      ctx.shadowColor = '#f80'; ctx.shadowBlur = 18;
      ctx.fillStyle = '#ff8a1e';
      ctx.beginPath();
      ctx.moveTo(h.r*1.9, 0); ctx.lineTo(-h.r, h.r*0.85); ctx.lineTo(-h.r*0.4, 0); ctx.lineTo(-h.r, -h.r*0.85);
      ctx.closePath(); ctx.fill();
      ctx.shadowBlur = 0; ctx.fillStyle = 'rgba(255,170,60,0.35)';
      ctx.fillRect(-h.r*3.5, -2, h.r*3, 4);
      ctx.restore();
      return;
    }
    ctx.rotate(h.rot);
    ctx.shadowColor = h.homing ? '#f0f' : '#f33'; ctx.shadowBlur = 15;
    ctx.fillStyle = h.homing ? '#c2f' : '#e22';
    ctx.beginPath();
    const spikes = 8;
    for (let i = 0; i < spikes*2; i++) {
      const rr = i % 2 === 0 ? h.r : h.r*0.55;
      const a = i * Math.PI / spikes;
      ctx.lineTo(Math.cos(a)*rr, Math.sin(a)*rr);
    }
    ctx.closePath(); ctx.fill();
    ctx.fillStyle = '#300';
    ctx.beginPath(); ctx.arc(0,0,h.r*0.25,0,Math.PI*2); ctx.fill();
    ctx.restore();
  }
  function drawWarningIcon(x, y, dart, alpha, hunter){
    ctx.save();
    ctx.globalAlpha = alpha;
    ctx.translate(x, y);
    ctx.shadowColor = dart ? '#f80' : hunter ? '#f0f' : '#f33'; ctx.shadowBlur = 15;
    ctx.fillStyle = dart ? '#ff9a2e' : hunter ? '#d4f' : '#f44';
    ctx.beginPath(); ctx.moveTo(0,-13); ctx.lineTo(12,9); ctx.lineTo(-12,9); ctx.closePath(); ctx.fill();
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#200'; ctx.font = 'bold 14px sans-serif'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.fillText('!', 0, 2);
    ctx.restore();
  }
  function drawWarning(p, t){
    if (p.dart) {
      ctx.save();
      ctx.beginPath(); ctx.rect(0, TOP, W, H - TOP); ctx.clip();
      const k = 1 - p.t / p.max;
      ctx.strokeStyle = `rgba(255,150,40,${0.2 + 0.5*k})`;
      ctx.lineWidth = 2 + k*2; ctx.setLineDash([8, 8]); ctx.lineDashOffset = -t*140;
      ctx.beginPath(); ctx.moveTo(p.x, p.y); ctx.lineTo(p.x + Math.cos(p.a)*1400, p.y + Math.sin(p.a)*1400); ctx.stroke();
      ctx.restore();
    }
    const x = clamp(p.x, 18, W-18), y = clamp(p.y, TOP + 16, H-18);
    const pulse = 0.5 + 0.5*Math.sin(t*25);
    drawWarningIcon(x, y, p.dart, 0.5 + 0.5*pulse, p.hunter);
  }
  function drawLaser(l, t){
    ctx.save();
    const FH = H - TOP;
    if (l.warn > 0) {
      const k = 1 - l.warn / 1.1;
      const pulse = 0.5 + 0.5*Math.sin(t*(20 + k*30));
      ctx.fillStyle = `rgba(255,60,60,${0.05 + 0.1*k})`;
      if (l.horiz) ctx.fillRect(0, l.pos - l.w/2, W, l.w); else ctx.fillRect(l.pos - l.w/2, TOP, l.w, FH);
      ctx.strokeStyle = `rgba(255,80,80,${0.35 + 0.5*pulse})`;
      ctx.lineWidth = 2; ctx.setLineDash([12, 8]); ctx.lineDashOffset = -t*80;
      ctx.beginPath();
      if (l.horiz) { ctx.moveTo(0, l.pos); ctx.lineTo(W, l.pos); } else { ctx.moveTo(l.pos, TOP); ctx.lineTo(l.pos, H); }
      ctx.stroke();
    } else {
      const a = Math.min(1, l.on / 0.15);
      ctx.globalAlpha = a;
      ctx.shadowColor = '#f33'; ctx.shadowBlur = 30;
      ctx.fillStyle = '#f33';
      const w = l.w * (0.8 + 0.2*Math.sin(t*60));
      if (l.horiz) ctx.fillRect(0, l.pos - w/2, W, w); else ctx.fillRect(l.pos - w/2, TOP, w, FH);
      ctx.shadowBlur = 0; ctx.fillStyle = '#fee';
      if (l.horiz) ctx.fillRect(0, l.pos - 3, W, 6); else ctx.fillRect(l.pos - 3, TOP, 6, FH);
    }
    ctx.restore();
  }
  function drawPower(p, t){
    const blink = p.life < 2 && Math.floor(p.life*8) % 2 === 0;
    if (blink) return;
    ctx.save();
    ctx.translate(p.x, p.y + Math.sin(p.t*4)*3);
    if (p.kind === 'life') {
      ctx.strokeStyle = 'rgba(120,255,150,0.6)'; ctx.lineWidth = 2;
      ctx.beginPath(); ctx.arc(0,0,18 + Math.sin(p.t*6)*2,0,Math.PI*2); ctx.stroke();
      ctx.shadowColor = '#4f8'; ctx.shadowBlur = 25;
      ctx.fillStyle = '#7f9';
      ctx.fillRect(-4, -11, 8, 22); ctx.fillRect(-11, -4, 22, 8);
      ctx.restore();
      return;
    }
    ctx.strokeStyle = 'rgba(255,227,107,0.6)'; ctx.lineWidth = 2;
    ctx.beginPath(); ctx.arc(0,0,18 + Math.sin(p.t*6)*2,0,Math.PI*2); ctx.stroke();
    ctx.rotate(p.t*2);
    ctx.shadowColor = '#fd4'; ctx.shadowBlur = 25;
    ctx.fillStyle = '#ffe36b';
    ctx.beginPath(); ctx.moveTo(0,-12); ctx.lineTo(9,0); ctx.lineTo(0,12); ctx.lineTo(-9,0); ctx.closePath(); ctx.fill();
    ctx.fillStyle = '#fff'; ctx.beginPath(); ctx.arc(0,0,3,0,Math.PI*2); ctx.fill();
    ctx.restore();
  }
  function drawPlayer(t){
    const fever = mult() === 4;
    for (let i = 0; i < trail.length; i++) {
      const a = i / trail.length;
      ctx.globalAlpha = a * (dashT > 0 ? 0.6 : fever ? 0.5 : 0.35);
      ctx.fillStyle = dashT > 0 ? '#fff2b0' : fever ? `hsl(${(t*300 + i*20) % 360},100%,65%)` : '#fc4';
      ctx.beginPath(); ctx.arc(trail[i].x, trail[i].y, player.r * a * 0.9, 0, Math.PI*2); ctx.fill();
    }
    ctx.globalAlpha = 1;
    if (dashT <= 0 && invuln > 0 && Math.floor(invuln*10) % 2 === 0) return;
    if (fever) glowCircle(player.x, player.y, player.r + 5 + Math.sin(t*10)*2, 'rgba(255,120,255,0.25)', 30);
    glowCircle(player.x, player.y, player.r, dashT > 0 ? '#fff0b0' : '#ffb627', dashT > 0 ? 40 : 28);
    glowCircle(player.x, player.y, player.r*0.55, '#fff6d8', 6);
    if (dashCd > 0) {
      ctx.save();
      ctx.strokeStyle = 'rgba(150,255,255,0.45)'; ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.arc(player.x, player.y, player.r + 4, -Math.PI/2, -Math.PI/2 + Math.PI*2*(1 - dashCd/DASH_CD));
      ctx.stroke();
      ctx.restore();
    }
    if (still > IDLE_LIMIT) {
      ctx.save();
      ctx.strokeStyle = `rgba(220,80,255,${0.4 + 0.4*Math.sin(t*12)})`; ctx.lineWidth = 2;
      ctx.beginPath(); ctx.arc(player.x, player.y, player.r + 14 + Math.sin(t*8)*3, 0, Math.PI*2); ctx.stroke();
      ctx.restore();
    }
    if (shield > 0 && !(shield < 1.2 && Math.floor(shield*10) % 2 === 0)) {
      ctx.save();
      ctx.translate(player.x, player.y); ctx.rotate(t*3);
      ctx.strokeStyle = '#ffe36b'; ctx.lineWidth = 3; ctx.shadowColor = '#fd4'; ctx.shadowBlur = 18;
      ctx.setLineDash([10, 6]);
      ctx.beginPath(); ctx.arc(0,0,player.r + 9,0,Math.PI*2); ctx.stroke();
      ctx.restore();
    }
  }

    ctx.font = `bold ${size}px system-ui,Segoe UI,Arial,sans-serif`;
    ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.fillStyle = color;
    if (alpha !== undefined) ctx.globalAlpha = alpha;
    if (glow) { ctx.shadowColor = glow; ctx.shadowBlur = 20; }
    ctx.fillText(txt, W/2, y);
    ctx.restore();
  }

  function draw(t){
    ctx.clearRect(0,0,W,H);
    ctx.save();
    if (shake > 0) ctx.translate(rand(-shake,shake), rand(-shake,shake));
    drawStars(t);
    drawGrid(t);

    if (state === 'menu') {
      for (const o of menuOrbs) glowCircle(o.x, o.y, 7 + Math.sin(o.t)*2, `hsl(${o.hue},100%,65%)`, 20);
    } else {
      if (state === 'playing') for (const l of lasers) if (l.warn > 0) drawLaser(l, t);
      if (state === 'playing') for (const p of pending) if (p.dart) drawWarning(p, t);
      for (const o of orbs) {
        if (o.life < 2 && Math.floor(o.life*8) % 2 === 0) continue;
        const grow = clamp(o.born * 4, 0, 1);
        if (grow <= 0) continue;
        const r = (o.r + Math.sin(o.t)*2) * grow;
        glowCircle(o.x, o.y, r, o.bonus ? `hsl(${o.hue},100%,62%)` : `hsl(${o.hue},100%,${o.big ? 75 : 64}%)`, o.big ? 35 : 26);
        glowCircle(o.x, o.y, r*0.42, '#fff', 5);
        ctx.strokeStyle = o.big ? `hsla(${o.hue},100%,80%,0.55)` : `hsla(${o.hue},100%,80%,0.35)`;
        ctx.lineWidth = o.big ? 2 : 1.5;
        ctx.beginPath(); ctx.arc(o.x, o.y, r + (o.big ? 6 : 5), -Math.PI/2, -Math.PI/2 + Math.PI*2*Math.max(0, o.life/o.max)); ctx.stroke();
      }
      if (power) drawPower(power, t);
      ctx.save();
      ctx.beginPath(); ctx.rect(-20, TOP, W + 40, H - TOP + 20); ctx.clip();
      if (state !== 'playing') ctx.globalAlpha = 0.55;
      for (const h of hazards) drawHazard(h);
      ctx.restore();
      if (state === 'playing') {
        for (const p of pending) if (!p.dart) drawWarning(p, t);
        drawPlayer(t);
        for (const l of lasers) if (l.warn <= 0) drawLaser(l, t);
      } else if (state === 'over' && player) {
        ctx.save();
